class OrderAuditLog(models.Model):
    """Immutable audit record that survives order deletion."""
    customer = models.ForeignKey(Customer, on_delete=models.CASCADE, related_name='order_audit_logs')
    order_uuid = models.UUIDField(db_index=True)
    checksum = models.CharField(max_length=64)
    actor_email = models.EmailField()
    recorded_at = models.DateTimeField()
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ['-recorded_at']
        indexes = [
            models.Index(fields=['customer', 'order_uuid']),
        ]

    def __str__(self):
        return f"AuditLog {self.pk} — {self.actor_email} @ {self.recorded_at}"

class TestAuditLogSurvivesOrderDeletion(TestCase):
    def setUp(self):
        self.customer = CustomerFactory()
        self.order = OrderFactory(customer=self.customer)
        self.log = OrderAuditLog.objects.create(
            customer=self.customer,
            order_uuid=self.order.uuid,
            checksum='a' * 64,
            actor_email='actor@example.com',
            recorded_at=timezone.now(),
        )

    def test_audit_log_survives_order_deletion(self):
        """Deleting the order must NOT cascade to audit logs."""
        self.order.delete()
        self.assertTrue(
            OrderAuditLog.objects.filter(pk=self.log.pk).exists()
        )
